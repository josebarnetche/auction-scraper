Python SDK Example

# subasto_client.py - Python SDK for Subasto API
import requests
from web3 import Web3
from typing import Optional, Dict, Any

class SubastoClient:
    """Subasto API client with USDC payment support."""

    BASE_URL = "https://subasto.com.ar/api/v1"
    USDC_ADDRESS = "0x833589fCD6eDb6E08f4c7C32D4f71b54bdA02913"
    PAYMENT_WALLET = "0x29E007249b744892a1da17F4289f75cfC871d6Fe"

    def __init__(self, private_key: Optional[str] = None, rpc_url: str = "https://mainnet.base.org"):
        self.w3 = Web3(Web3.HTTPProvider(rpc_url))
        self.private_key = private_key
        if private_key:
            self.account = self.w3.eth.account.from_key(private_key)

    def get_pricing(self) -> Dict[str, Any]:
        """Get API pricing (free endpoint)."""
        response = requests.get(f"{self.BASE_URL}/price")
        return response.json()

    def send_payment(self, amount_usdc: float) -> str:
        """Send USDC payment and return transaction hash."""
        if not self.private_key:
            raise ValueError("Private key required for payments")

        # USDC has 6 decimals
        amount_wei = int(amount_usdc * 10**6)

        # ERC20 transfer ABI
        usdc_abi = [{"inputs": [{"name": "to", "type": "address"},
                    {"name": "amount", "type": "uint256"}],
                    "name": "transfer", "outputs": [{"type": "bool"}], "type": "function"}]

        usdc = self.w3.eth.contract(address=self.USDC_ADDRESS, abi=usdc_abi)

        tx = usdc.functions.transfer(self.PAYMENT_WALLET, amount_wei).build_transaction({
            'from': self.account.address,
            'nonce': self.w3.eth.get_transaction_count(self.account.address),
            'gas': 100000,
            'maxFeePerGas': self.w3.eth.gas_price * 2,
            'maxPriorityFeePerGas': self.w3.to_wei(0.001, 'gwei')
        })

        signed = self.w3.eth.account.sign_transaction(tx, self.private_key)
        tx_hash = self.w3.eth.send_raw_transaction(signed.rawTransaction)

        # Wait for confirmation
        self.w3.eth.wait_for_transaction_receipt(tx_hash)
        return tx_hash.hex()

    def get_premium(self, tx_hash: str) -> Dict[str, Any]:
        """Get premium picks ($0.01 USDC)."""
        response = requests.get(f"{self.BASE_URL}/premium", params={"tx": tx_hash})
        return response.json()

    def get_opportunities(self, tx_hash: str) -> Dict[str, Any]:
        """Get hot deals ($0.02 USDC)."""
        response = requests.get(f"{self.BASE_URL}/opportunities", params={"tx": tx_hash})
        return response.json()

    def get_auctions(self, tx_hash: str) -> Dict[str, Any]:
        """Get all auctions ($0.05 USDC)."""
        response = requests.get(f"{self.BASE_URL}/auctions", params={"tx": tx_hash})
        return response.json()

JavaScript/Node SDK Example

// subasto-client.js - Node.js SDK for Subasto API
import { createPublicClient, createWalletClient, http, parseAbi } from 'viem';
import { base } from 'viem/chains';
import { privateKeyToAccount } from 'viem/accounts';

const BASE_URL = 'https://subasto.com.ar/api/v1';
const USDC_ADDRESS = '0x833589fCD6eDb6E08f4c7C32D4f71b54bdA02913';
const PAYMENT_WALLET = '0x29E007249b744892a1da17F4289f75cfC871d6Fe';

class SubastoClient {
  constructor(privateKey = null) {
    this.publicClient = createPublicClient({
      chain: base,
      transport: http()
    });

    if (privateKey) {
      this.account = privateKeyToAccount(privateKey);
      this.walletClient = createWalletClient({
        account: this.account,
        chain: base,
        transport: http()
      });
    }
  }

  async getPricing() {
    const response = await fetch(`${BASE_URL}/price`);
    return response.json();
  }

  async sendPayment(amountUsdc) {
    if (!this.walletClient) {
      throw new Error('Private key required for payments');
    }

    const amountWei = BigInt(Math.floor(amountUsdc * 1e6));

    const hash = await this.walletClient.writeContract({
      address: USDC_ADDRESS,
      abi: parseAbi(['function transfer(address to, uint256 amount) returns (bool)']),
      functionName: 'transfer',
      args: [PAYMENT_WALLET, amountWei]
    });

    // Wait for confirmation
    await this.publicClient.waitForTransactionReceipt({ hash });
    return hash;
  }

  async getPremium(txHash) {
    const response = await fetch(`${BASE_URL}/premium?tx=${txHash}`);
    return response.json();
  }

  async getOpportunities(txHash) {
    const response = await fetch(`${BASE_URL}/opportunities?tx=${txHash}`);
    return response.json();
  }

  async getAuctions(txHash) {
    const response = await fetch(`${BASE_URL}/auctions?tx=${txHash}`);
    return response.json();
  }
}

// Usage example
const client = new SubastoClient();
const pricing = await client.getPricing();
console.log('Payment wallet:', pricing.payment.wallet);

// With wallet (automated)

// const txHash = await client.sendPayment(0.01);
// const data = await client.getPremium(txHash);

export default SubastoClient;

Use Case: Auction Monitoring Bot

Build a bot that monitors auctions and alerts on new listings matching your criteria:

import time
import json
from subasto_client import SubastoClient

class AuctionMonitor:
    def __init__(self, private_key):
        self.client = SubastoClient(private_key)
        self.seen_ids = set()
        self.filters = {
            'category': 'vehicles',
            'min_discount': 50,  # 50%+ off
            'max_price_usd': 10000
        }

    def check_new_opportunities(self):
        # Pay for opportunities endpoint
        tx_hash = self.client.send_payment(0.02)
        data = self.client.get_opportunities(tx_hash)

        new_matches = []
        for opp in data['data']['opportunities']:
            listing = opp['listing']

            # Skip if already seen
            if listing['id'] in self.seen_ids:
                continue

            # Apply filters
            if listing['category'] != self.filters['category']:
                continue
            if opp['discount_percentage'] < self.filters['min_discount']:
                continue
            if opp['auction_price_usd'] > self.filters['max_price_usd']:
                continue

            self.seen_ids.add(listing['id'])
            new_matches.append(opp)

        return new_matches

    def send_alert(self, opportunities):
        for opp in opportunities:
            print(f"""
NEW DEAL ALERT!
Title: {opp['listing']['title']}
Discount: {opp['discount_percentage']}%
Price: ${opp['auction_price_usd']:.0f} (Market: ${opp['market_median_usd']:.0f})
URL: {opp['listing']['source_url']}
""")

    def run(self, interval_hours=24):
        while True:
            new_opps = self.check_new_opportunities()
            if new_opps:
                self.send_alert(new_opps)
            time.sleep(interval_hours * 3600)

# Run the monitor
monitor = AuctionMonitor(private_key="0x...")
monitor.run(interval_hours=24)  # Check daily ($0.02/day)

def analyze_investment_opportunities(subasto_key, anthropic_key):

Use Case: Arbitrage Detection

Detect arbitrage opportunities between auction and market prices:

from subasto_client import SubastoClient
import pandas as pd

def find_arbitrage_opportunities(client, min_profit_usd=1000):
    tx_hash = client.send_payment(0.02)
    data = client.get_opportunities(tx_hash)

    opportunities = []
    for opp in data['data']['opportunities']:
        # Calculate potential profit
        buy_price = opp['auction_price_usd']
        market_price = opp['market_median_usd']

        # Assume 15% selling costs (fees, taxes, transport)
        net_sell_price = market_price * 0.85
        profit = net_sell_price - buy_price
        roi = (profit / buy_price) * 100 if buy_price > 0 else 0

        if profit >= min_profit_usd and roi >= 30:
            opportunities.append({
                'id': opp['listing']['id'],
                'title': opp['listing']['title'],
                'category': opp['listing']['category'],
                'buy_price_usd': buy_price,
                'market_price_usd': market_price,
                'net_profit_usd': profit,
                'roi_percent': roi,
                'url': opp['listing']['source_url'],
                'confidence': opp['confidence_score']
            })

    # Sort by ROI
    df = pd.DataFrame(opportunities)
    return df.sort_values('roi_percent', ascending=False)

# Find opportunities with $1000+ profit potential
client = SubastoClient(private_key="0x...")
arb_opps = find_arbitrage_opportunities(client, min_profit_usd=1000)
print(arb_opps.to_string())
